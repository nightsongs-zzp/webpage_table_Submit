// 移动端汉堡菜单 + 页头高度发布
(function () {
var btn = document.getElementById('menuBtn');
var menu = document.getElementById('mobileMenu');
var iOpen = document.getElementById('menuIconOpen');
var iClose = document.getElementById('menuIconClose');
var header = menu ? menu.parentElement : null;
// 把页头实际高度写进 --exp-header-h。目录抽屉原先硬编码 top:4rem（假定页头恰好 64px），
// 但下拉菜单展开后页头会变高，硬编码值会把抽屉盖到菜单上。
function syncHeaderHeight() {
if (header && header.offsetHeight) {
document.documentElement.style.setProperty('--exp-header-h', header.offsetHeight + 'px');
}
}
syncHeaderHeight();
window.addEventListener('resize', syncHeaderHeight);
if (window.ResizeObserver && header) new ResizeObserver(syncHeaderHeight).observe(header);
if (!btn || !menu) return;
btn.addEventListener('click', function () {
var nowHidden = menu.classList.toggle('hidden');
btn.setAttribute('aria-expanded', nowHidden ? 'false' : 'true');
btn.setAttribute('aria-label', nowHidden ? '打开菜单' : '关闭菜单');
if (iOpen) iOpen.style.display = nowHidden ? '' : 'none';
if (iClose) iClose.style.display = nowHidden ? 'none' : '';
});
})();
</script>

// 点击统计埋点（实验详情页）
window.__PAGE_KEY__ = "physics-lab/experiment";
(function () {
var ENDPOINT = "/api/event";
function send(action, payload) {
try {
var body = Object.assign({ action: action, page_key: window.__PAGE_KEY__ }, payload);
navigator.sendBeacon
? navigator.sendBeacon(ENDPOINT, new Blob([JSON.stringify(body)], { type: "application/json" }))
: fetch(ENDPOINT, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), keepalive: true });
} catch (e) {}
}

function norm(s) { return String(s || "").trim(); }

function bind(id, name, category) {
var el = document.getElementById(id);
if (!el) return;
el.setAttribute("data-click-key", "btn:" + id);
el.setAttribute("data-click-name", name || id);
el.setAttribute("data-click-category", category || "实验详情");
send("register", { btn_key: "btn:" + id, btn_name: norm(name || id), category: norm(category || "实验详情") });
}

document.addEventListener("DOMContentLoaded", function () {
bind("favoriteBtn", "收藏实验", "实验详情-操作");
bind("toggleSidebarBtn", "关闭目录", "实验详情-目录");
bind("sidebarToggleBtnFixed", "打开目录", "实验详情-目录");

// 目录项
document.querySelectorAll(".catalog-item").forEach(function (item) {
var t = norm(item.textContent);
var key = "catalog:" + norm(item.getAttribute("data-target") || t);
item.setAttribute("data-click-key", key);
item.setAttribute("data-click-name", t);
item.setAttribute("data-click-category", "实验详情-目录");
send("register", { btn_key: key, btn_name: t, category: "实验详情-目录" });
});

// 评论相关
document.querySelectorAll("button[onclick='postComment()']").forEach(function (btn) {
btn.setAttribute("data-click-key", "comment:post");
btn.setAttribute("data-click-name", "发布评论");
btn.setAttribute("data-click-category", "实验详情-评论");
send("register", { btn_key: "comment:post", btn_name: "发布评论", category: "实验详情-评论" });
});
});

document.addEventListener("click", function (e) {
var el = e.target && e.target.closest ? e.target.closest("[data-click-key]") : null;
if (!el) return;
var key = el.getAttribute("data-click-key");
var name = el.getAttribute("data-click-name") || el.innerText || el.textContent || "";
var cat = el.getAttribute("data-click-category") || "实验详情";
send("click", { btn_key: norm(key).slice(0, 120), btn_name: norm(name).slice(0, 200), category: norm(cat).slice(0, 80) });
}, true);
})();
